======
RULES
======

[ > player | no activetile ] -> cancel

horizontal [> verplayer | ] -> [ | ortplayer ]
horizontal [> ortplayer | ] -> [ | horplayer ]
horizontal [> horplayer | ] -> [ | horplayer ]

vertical [> horplayer | ] -> [ | ortplayer ]
vertical [> ortplayer | ] -> [ | verplayer ]
vertical [> verplayer | ] -> [ | verplayer ]

[ activetile | ortplayer ] -> [ temp | ortplayer ]
[ no temp | ortplayer ] -> [ activetile | ortplayer ]

vertical [ activetile | horplayer ] -> [ temp | horplayer ]
vertical [ no temp | horplayer] -> [ activetile | horplayer ]

horizontal [ activetile | verplayer ] -> [ temp | verplayer ]
horizontal [ no temp | verplayer] -> [ activetile | verplayer ]

[ temp ] -> []

late [activetile void ] -> [ void ]
