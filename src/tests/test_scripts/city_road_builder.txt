======
RULES
======

(clearing)
[Cantmove no Path] -> []
[Cantwin] -> []

[WillWin] -> Win

(check win conditions)
[Player] [CornerWonrg] -> [Player CantWin] [CornerWonrg]
[Player] [HouseNeut] -> [Player CantWin] [HouseNeut]
[Player] [HouseWonrg] -> [Player CantWin] [HouseWonrg]
up [up Player no CantWin|Startpos] -> [PathU|PathD PlayerDone WillWin]
down [down Player no CantWin|Startpos] -> [PathD|PathU PlayerDone WillWin]
left [left Player no CantWin|Startpos] -> [PathL|PathR PlayerDone WillWin]
right [right Player no CantWin|Startpos] -> [PathR|PathL PlayerDone WillWin]
[WillWin] -> again

(moves)
(forward move)
right [right Player|no Wall no Cantmove] -> [PathR|PathL Player] sfx0
right [no Wall no Cantmove|left Player] -> [Player PathR|PathL] sfx0
down [down Player|no Wall no Cantmove] -> [PathD|PathU Player] sfx0
down [no Wall no Cantmove|up Player] -> [Player PathD|PathU] sfx0
(backwards undo moves)
right [> Player PathR | PathL ] -> [|Player] sfx1
down [> Player PathD | PathU ] -> [|Player] sfx1
left [> Player PathL | PathR ] -> [|Player] sfx1
up [> Player PathU | PathD ] -> [|Player] sfx1

(VERY IMPORTANT)
[> Player] -> [Player]
[Path] -> [Path Cantmove]

(paths)
(straight)
late [PathL PathR] -> [PathL PathR Straight]
late [PathU PathD] -> [PathU PathD Straight]
(corner)
late [PathL PathU] -> [PathL PathU Corner]
late [PathU PathR] -> [PathU PathR Corner]
late [PathR PathD] -> [PathR PathD Corner]
late [PathD PathL] -> [PathD PathL Corner]
(can't turn on empty)
late [Corner no House no Intersect no CornerWonrg] -> [Corner CornerWonrg] sfx3
[CornerWonrg no Corner] -> [] sfx4
(clearing)
late [Straight PathL no PathR] -> [PathL] again
late [Straight PathR no PathL] -> [PathR] again
late [Straight PathU no PathD] -> [PathU] again
late [Straight PathD no PathU] -> [PathD] again

late [Corner PathL no PathU no PathD] -> [PathL] again
late [Corner PathR no PathU no PathD] -> [PathR] again
late [Corner PathU no PathL no PathR] -> [PathU] again
late [Corner PathD no PathL no PathR] -> [PathD] again

(turning)
late [Corner HouseNeut] -> [Corner HouseRight] sfx2
late [Straight HouseNeut] -> [Straight HouseWonrg] sfx3
[HouseRight no Corner] -> [HouseNeut] sfx4
[HouseWonrg no Straight] -> [HouseNeut] sfx4
