RULES 
====== 

( gun charging )
[GunOrGunBroken] -> [GunOrGunBroken TempGun]
[Gun0 TempGun] -> [Gun1 no TempGun]
[Gun1 TempGun] -> [Gun2 no TempGun]
[Gun2 TempGun] -> [Gun3 no TempGun]
[Gun3 TempGun] -> [Gun4 no TempGun]
right [Gun4 TempGun | no ObjectR] -> [Gun0 BulletR no TempGun | ]
right [Gun4 TempGun | ObjectR] -> [GunBroken1 no TempGun | ObjectR] again
( gun breaking )
[GunBroken1 TempGun] -> [GunBroken2 no TempGun] again
[GunBroken2 TempGun] -> [GunBroken3 no TempGun] again
[GunBroken3 TempGun] -> [GunBroken4 no TempGun] again
[GunBroken4 TempGun] -> [GunBroken5 no TempGun] again
[GunBroken5 TempGun] -> [GunBroken6 no TempGun] again
[GunBroken6 TempGun] -> [GunBroken7 no TempGun] again
[GunBroken7 TempGun] -> [GunBroken8 no TempGun] win

( bullet moving )
[Bullet BorderCheck] -> [Player TempTele BorderCheck]
[BulletD][Gun] -> [down BulletD][Gun]
[BulletL][Gun] -> [left BulletL][Gun]
[BulletR][Gun] -> [right BulletR][Gun]
[BulletU][Gun] -> [up BulletU][Gun]
( bullet and object colliding )
down [> BulletD | ObjectD] -> [stationary Player < TempTele | ObjectD]
late [ BulletD CrateUL] -> [ BulletL CrateUL]
late [ BulletD CrateUR] -> [ BulletR CrateUR]
left [> BulletL | ObjectL] -> [stationary Player < TempTele | ObjectL]
late [ BulletL CrateUR] -> [ BulletU CrateUR]
late [ BulletL CrateDR] -> [ BulletD CrateDR]
right [> BulletR | ObjectR] -> [stationary Player < TempTele | ObjectR]
late [ BulletR CrateUL] -> [BulletU CrateUL]
late [ BulletR CrateDL] -> [BulletD CrateDL]
up [> BulletU | ObjectU] -> [stationary Player < TempTele | ObjectU]
late [BulletU CrateDL] -> [ BulletL CrateDL]
late [BulletU CrateDR] -> [ BulletR CrateDR]
( teleport collision recoiling )
[> TempTele Object] -> [> TempTele > Object]

( teleportation happening )
[Player no TempTele][TempTele] -> [][TempTele]
[DeadPlayer no TempTele][TempTele] -> [] [TempTele]
[TempTele] -> []

( player, object normally moving )
[> Player | Object] -> [> Player | > Object]
[> Object | stationary Object] -> [> Object | > Object]
( cannot push gun )
[> Player | GunOrGunBroken] -> [stationary Player | GunOrGunBroken]
[> Object | GunOrGunBroken] -> [stationary Object | GunOrGunBroken]
[> Object | stationary Object] -> [stationary Object | Object]
[> Player | stationary Object] -> [stationary Player | Object]
( pit )
late [Player Pit] -> [DeadPlayer Pit]
[> Object | Pit] -> [ | Pit]
( checkpoint )
late [SolutionCheck Bullet][SaveFinal] -> [Bullet][Save]
late [Player Save] -> [Player] checkpoint
