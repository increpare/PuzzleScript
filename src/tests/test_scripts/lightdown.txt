======
RULES
======

(Clearing)
[Light] -> []
[TargetActi] -> [Target]

(Basic movement)
[> Player | Crate] -> [> Player | > Crate]
[> Player | Wall] -> [Player | Wall]

[> Player] [> Crate | Crate] -> [Player] [Crate | Crate]
[> Player] [> Crate | Wall] -> [Player] [Crate | Wall]
[> Crate FakeCrate | FakeCrate] -> [ | Crate FakeCrate] sfx0

[> Crate FakeCrate] -> [> Crate > FakeCrate]

late [Target Crate] -> [TargetActi Crate]
late [Target FakeCrate] -> [TargetActi FakeCrate]

(Lighting)
late [LightCrate] -> [LightFull LightCrate]
+ late [LightWall] -> [LightFull LightWall]
+ late [LightFull | no Light no SolidWall] -> [LightFull | LightMid]
+ late [LightMid | no Light no SolidWall] -> [LightMid | LightFew]

(Darkess effects)
late [NormalWall no Light] -> []
late [FakeWall Light] -> [FakeWall NormalWall Light]

late [NormalCrate no Light] -> []
late [FakeCrate Light] -> [FakeCrate NormalCrate Light]
