======
RULES
======
(death and win)

late [ player death ] -> [falldeath1 death] sfx1 again

Late [player killing] -> [deathani1 killing] sfx2 again
[player queen] -> sfx0

[> player target] -> win


(animation)
[stationary Player] [poof1] -> [stationary Player] []
[stationary Player] [poof] -> [stationary Player] [poof1]

[stationary Player] [queen2 no temp ] -> [stationary Player] [queen3 temp]
[stationary Player] [queen3 no temp ] -> [stationary Player] [queen2 temp]


[deathani1 no temp ] -> [deathani2 temp] again
[deathani2 no temp ] -> [deathani3 temp] again
[deathani3 no temp ] -> [deathani4 temp] again
[deathani4 no temp ] -> [deathani5 temp] again
[deathani5 no temp ] -> restart

[falldeath1 no temp ] -> [falldeath2 temp] again
[falldeath2 no temp ] -> [falldeath3 temp] again
[falldeath3 no temp ] -> [falldeath4 temp] again
[falldeath4 no temp ] -> [falldeath5 temp] again
[falldeath5 no temp ] -> restart

[stationary Player] [fire1 no temp ] -> [stationary Player] [fire2 temp]
[stationary Player] [fire2 no temp ] -> [stationary Player] [fire3 temp]
[stationary Player] [fire3 no temp ] -> [stationary Player] [fire1 temp]

late [temp] -> []

right [right PlayerFalling |no ground] -> [ | down PlayerRight]
left [left PlayerFalling |no ground] -> [ | down Playerleft]
right [right PlayerFalling | ground] -> [ down PlayerRight | ground]
left [left PlayerFalling | ground] -> [ down Playerleft | ground]

(falling)
[ moving Player ] [PlayerFalling | No ground] -> [ moving Player ] [down PlayerFalling | ] 
random down [PlayerFalling | ground ] -> [PlayerGround | ground ]
down [PlayerGround | No ground] -> [down PlayerFall | ]

down [ moving Player ] [PlayerFalling | celing] -> [ moving Player ] [PlayerGround | celing]

[PlayerJumpStart] -> [PlayerJump]

up [playerjumping | Item ] -> [PlayerFall | Item]


up [PlayerJump1 | No Item] -> [ | PlayerFall] 
up [PlayerJump | No Item] -> [ | PlayerJump1] again
up [PlayerJumpStart | No Item] -> [ | PlayerJump] again
up [ up PlayerGround | No Item] -> [ | PlayerJumpStart] sfx3 again

(crates)
(push)
horizontal [ moving Player ] [> Player | Crate] -> [ moving Player ] [> Player | > Crate]
[> crate |crate] -> [> crate | > Crate]

(fire!)
Horizontal [woodcrate |fire] -> [poof | fire] sfx4 
Horizontal [poof1 |fire] -> [fire1| fire] sfx4
vertical [woodcrate |fire] -> [poof |fire] sfx4 again
vertical [metalcrate |fire] -> [down metalcrate |poof] sfx4 again
[metalcrate fire] -> [metalcrate] sfx4 again

(falling)
down [Crate | No ground] -> [down Crate | ] again

down [fire | No ground] -> [down fire | ] again
down [fire | fire] -> [ |fire ] again

WINCONDITIONS

any player on end
