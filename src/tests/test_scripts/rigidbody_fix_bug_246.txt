======
RULES
======

(Can't move into beams)
Horizontal [> Player | HBeam] -> cancel
Vertical [> Player | VBeam] -> cancel

(Update whether everything is active or inactive)
[> IPlayer | IStart] -> [> IPlayer AMarker | IStart]
late [IPlayer] [AMarker] -> [APlayer] [AMarker]
late [IProjector] [AMarker] -> [AProjector] [AMarker]
late [IOBlock] [AMarker] -> [AOBlock] [AMarker]
late [ILBlock] [AMarker] -> [ALBlock] [AMarker]
late [IRBlock] [AMarker] -> [ARBlock] [AMarker]
(late [IHBlock] [AMarker] -> [AHBlock] [AMarker]
late [IVBlock] [AMarker] -> [AVBlock] [AMarker])
late [IUBlock] [AMarker] -> [AUBlock] [AMarker]
late [IDBlock] [AMarker] -> [ADBlock] [AMarker]
late [IWall] [AMarker] -> [AWall] [AMarker]
late [IBackground] [AMarker] -> [ABackground] [AMarker]
late [IGoal] [AMarker] -> [AGoal] [AMarker]
late [IStart] [AMarker] -> [AStart] [AMarker]
late [AMarker] -> []

[> APlayer | Beam] -> [> APlayer IMarker | Beam]
late [APlayer] [IMarker] -> [IPlayer] [IMarker]
late [AProjector] [IMarker] -> [IProjector] [IMarker]
late [AOBlock] [IMarker] -> [IOBlock] [IMarker]
late [ALBlock] [IMarker] -> [ILBlock] [IMarker]
late [ARBlock] [IMarker] -> [IRBlock] [IMarker]
(late [AHBlock] [IMarker] -> [IHBlock] [IMarker]
late [AVBlock] [IMarker] -> [IVBlock] [IMarker])
late [AUBlock] [IMarker] -> [IUBlock] [IMarker]
late [ADBlock] [IMarker] -> [IDBlock] [IMarker]
late [AWall] [IMarker] -> [IWall] [IMarker]
late [ABackground] [IMarker] -> [IBackground] [IMarker]
late [AGoal] [IMarker] -> [IGoal] [IMarker]
late [AStart] [IMarker] -> [IStart] [IMarker]
late [IMarker] -> []

(Player Push Blocks)
rigid [> Player | Block] -> [> Player | > Block]
+ rigid [> Block | Block] -> [> Block | > Block]
+ rigid Left [Moving RBlock | LBlock] -> [Moving RBlock | Moving LBlock]
+ rigid Right [Moving LBlock | RBlock] -> [Moving LBlock | Moving RBlock]
+ rigid Up [Moving DBlock | UBlock] -> [Moving DBlock | Moving UBlock]
+ rigid Down [Moving UBlock | DBlock] -> [Moving UBlock | Moving DBlock]
(+ Vertical [Moving VBlock | VUDBlock] -> [Moving VBlock | Moving VUDBlock])

(Beams Push)
late [MMarker] -> []

rigid Right [RPusher | Pushable] [TMarker] [Player] -> [RPusher | > Pushable] [TMarker] [Player MMarker] again
+ rigid Right [> Pushable | Pushable] -> [> Pushable | > Pushable]
+ rigid Left [Moving RBlock | LBlock] -> [Moving RBlock | Moving LBlock]
+ rigid Right [Moving LBlock | RBlock] -> [Moving LBlock | Moving RBlock]
+ rigid Up [Moving DBlock | UBlock] -> [Moving DBlock | Moving UBlock]
+ rigid Down [Moving UBlock | DBlock] -> [Moving UBlock | Moving DBlock]

rigid Left [LPusher | Pushable] [TMarker] [Player] -> [LPusher | > Pushable] [TMarker] [Player MMarker] again
+ rigid Left [> Pushable | Pushable] -> [> Pushable | > Pushable]
+ rigid Left [Moving RBlock | LBlock] -> [Moving RBlock | Moving LBlock]
+ rigid Right [Moving LBlock | RBlock] -> [Moving LBlock | Moving RBlock]
+ rigid Up [Moving DBlock | UBlock] -> [Moving DBlock | Moving UBlock]
+ rigid Down [Moving UBlock | DBlock] -> [Moving UBlock | Moving DBlock]

rigid Up [UPusher | Pushable] [TMarker] [Player] -> [UPusher | > Pushable] [TMarker] [Player MMarker] again
+ rigid Up [> Pushable | Pushable] -> [> Pushable | > Pushable]
+ rigid Left [Moving RBlock | LBlock] -> [Moving RBlock | Moving LBlock]
+ rigid Right [Moving LBlock | RBlock] -> [Moving LBlock | Moving RBlock]
+ rigid Up [Moving DBlock | UBlock] -> [Moving DBlock | Moving UBlock]
+ rigid Down [Moving UBlock | DBlock] -> [Moving UBlock | Moving DBlock]

rigid Down [DPusher | Pushable] [TMarker] [Player] -> [DPusher | > Pushable] [TMarker] [Player MMarker] again
+ rigid Down [> Pushable | Pushable] -> [> Pushable | > Pushable]
+ rigid Left [Moving RBlock | LBlock] -> [Moving RBlock | Moving LBlock]
+ rigid Right [Moving LBlock | RBlock] -> [Moving LBlock | Moving RBlock]
+ rigid Up [Moving DBlock | UBlock] -> [Moving DBlock | Moving UBlock]
+ rigid Down [Moving UBlock | DBlock] -> [Moving UBlock | Moving DBlock]


(beams detect motion)
Horizontal [ > Pushable | VBeam ] -> [ > Pushable | > VBeam ]
Vertical [ > Pushable | HBeam ] -> [ > Pushable | > HBeam ]

(If beams detected motion, transfer that to the projectors)
[ Projector | ... | Perpendicular Beam ] -> [ Perpendicular Projector | ... | Perpendicular Beam ]

(remove beams)
[ Beam ] -> []

(Place Beams)
late Left [ AProjector | No Blocker ] -> [ AProjector | ALBeam ]
late Left [ IProjector | No Blocker ] -> [ IProjector | ILBeam ]
late Right [ AProjector | No Blocker ] -> [ AProjector | ARBeam ]
late Right [ IProjector | No Blocker ] -> [ IProjector | IRBeam ]
late Horizontal [ HBeam | No Blocker No Projector No HBeam] -> [HBeam | HBeam]
late Up [ AProjector | No Blocker ] -> [ AProjector | AUBeam ]
late Up [ IProjector | No Blocker ] -> [ IProjector | IUBeam ]
late Down [ AProjector | No Blocker ] -> [ AProjector | ADBeam ]
late Down [ IProjector | No Blocker ] -> [ IProjector | IDBeam ]
late Vertical [ VBeam | No Blocker No Projector No VBeam] -> [VBeam | VBeam]

(Run a second turn)
[Moving Player No TMarker] -> [Moving Player TMarker] again
[Stationary Player No MMarker] [TMarker] -> [Player] []
